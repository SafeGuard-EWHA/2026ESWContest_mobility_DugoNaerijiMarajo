cmake_minimum_required(VERSION 3.16)
project(SafeGuardMain VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json 3.9 QUIET)
if(NOT nlohmann_json_FOUND)
  find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
  add_library(nlohmann_json INTERFACE)
  target_include_directories(nlohmann_json INTERFACE ${JSON_INCLUDE_DIR})
  add_library(nlohmann_json::nlohmann_json ALIAS nlohmann_json)
endif()
add_library(safeguard_core src/core.cpp)
target_include_directories(safeguard_core PUBLIC src)
target_link_libraries(safeguard_core PUBLIC nlohmann_json::nlohmann_json)
target_compile_options(safeguard_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(safeguard_main src/main.cpp)
target_link_libraries(safeguard_main PRIVATE safeguard_core)
target_compile_options(safeguard_main PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_core tests/test_core.cpp)
  target_link_libraries(test_core PRIVATE safeguard_core)
  target_compile_options(test_core PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME core COMMAND test_core)
  add_executable(test_input tests/test_input.cpp)
  target_link_libraries(test_input PRIVATE safeguard_core)
  target_compile_options(test_input PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME input_contract_and_rearm COMMAND test_input)
  add_executable(test_observer tests/test_observer.cpp)
  target_link_libraries(test_observer PRIVATE safeguard_core)
  target_compile_options(test_observer PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME thermal_observer COMMAND test_observer)
  add_executable(test_ambient_observer tests/test_ambient_observer.cpp)
  target_link_libraries(test_ambient_observer PRIVATE safeguard_core)
  target_compile_options(test_ambient_observer PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME ambient_observer COMMAND test_ambient_observer)
endif()

add_executable(test_predictive_control tests/test_predictive_control.cpp)
target_link_libraries(test_predictive_control PRIVATE safeguard_core)
target_compile_options(test_predictive_control PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
add_test(NAME predictive_control COMMAND test_predictive_control)
